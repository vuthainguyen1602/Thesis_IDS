\section{I.\ VẤN ĐỀ VÀ ĐÓNG GÓP}
% =====================================================================

% ---------------------------------------------------------------- 0:45
\begin{frame}{F1 gần như tuyệt đối trên CICIDS2017: cần kiểm chứng}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Hai câu hỏi}
      \begin{enumerate}
        \item \key{Có cần đủ $\approx$80 đặc trưng không?}\\
              {\small Càng nhiều đặc trưng càng tốn tính toán, khó chạy trên
              thiết bị biên.}
        \item \key{Điểm cao có đáng tin không?}\\
              {\small Hai lỗi khiến điểm cao hơn thực tế:}
              \begin{itemize}\small
                \item Cổng đích: mỗi loại tấn công nhắm vào một số cổng cố định,
                      nên \bad{cổng gần như tiết lộ nhãn}.
                \item Luồng trùng nhau nằm ở \emph{cả} tập huấn luyện
                      \emph{lẫn} tập kiểm tra: mô hình \bad{đã gặp dữ liệu kiểm tra từ trước}.
              \end{itemize}
      \end{enumerate}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{alertblock}{Bài báo này}
        Sửa cách đánh giá cho đúng \emph{trước}, rồi mới so sánh các cách
        giảm đặc trưng.
      \end{alertblock}
      \begin{exampleblock}{Thông điệp}
        \key{Đo thế nào} quan trọng không kém \emph{phương pháp nào thắng}.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{\say Bài báo xuất phát từ hai câu hỏi. Thứ nhất, mỗi luồng mạng có
  khoảng 80 đặc trưng, có cần dùng hết không, vì càng nhiều thì càng tốn và
  càng khó chạy trên thiết bị biên. Thứ hai, rất nhiều bài đạt F1 trên 0,99 với
  bộ dữ liệu này, điểm đó có đáng tin không. Có hai lỗi khiến điểm cao hơn thực tế. Một
  là cổng đích: trong bộ dữ liệu, mỗi loại tấn công nhắm vào một số cổng cố định, nên cổng gần như tiết lộ nhãn. Hai là luồng trùng nhau nằm ở cả tập huấn luyện
  lẫn tập kiểm tra, mô hình xem như đã gặp dữ liệu kiểm tra từ trước. Vì vậy chúng tôi sửa cách
  đánh giá trước, rồi mới so sánh các cách giảm đặc trưng.
  \ifasked Rò rỉ dữ liệu là khi thông tin không có lúc triển khai lọt vào huấn
  luyện hoặc đánh giá. Cột cổng đích trong dữ liệu là
  \texttt{destination\_port}.}
\end{frame}

% ---------------------------------------------------------------- 0:35
\begin{frame}{Các đóng góp}
  \begin{enumerate}
    \setlength{\itemsep}{0.35em}
    \item So sánh \key{4 cách giảm đặc trưng} (giữ hết, RF Importance, SHAP,
          PCA) trên \emph{cùng} 8 thuật toán Spark ML.
    \item \key{Quy trình đánh giá chống rò rỉ}: bỏ cổng, bỏ luồng trùng trước
          khi chia, không dùng tập kiểm tra để chọn bất cứ thứ gì.
    \item \key{Kiểm định thống kê} thay vì chỉ so sánh điểm số.
    \item Đánh giá những gì điểm tổng thường che khuất: \key{từng loại tấn công}, và
          khi \key{đổi sang bộ dữ liệu khác}.
    \item Chọn cấu hình \key{hợp với thiết bị biên} Jetson Orin Nano Super.
  \end{enumerate}
  \note{\say Bài báo có năm đóng góp. Một, so sánh bốn cách giảm đặc trưng
  trên cùng tám thuật toán, cùng một quy trình. Hai, quy trình đánh giá chống
  rò rỉ. Ba, dùng kiểm định thống kê để kết luận, không chỉ so sánh điểm số. Bốn,
  đánh giá những gì điểm tổng thường che khuất: từng loại tấn công, và khi đổi sang bộ
  dữ liệu khác. Năm, chọn cấu hình hợp với thiết bị biên. Điểm đáng nói là mục
  hai đến bốn ít khi được làm cùng nhau trong các bài về CICIDS2017.}
\end{frame}

\begin{frame}{Quy trình: mỗi bước chặn một đường rò rỉ}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \resizebox{0.98\textwidth}{!}{%
      \begin{tikzpicture}[

        \node[box, below=of prep] (split) {Chia 80\% huấn luyện / 20\% kiểm tra};
        \node[box, below=of split] (strat) {\textbf{4 cách giảm đặc trưng}: giữ hết $|$ RF $|$ SHAP $|$ PCA};
        \node[box, below=of strat] (clf) {8 thuật toán Spark ML $+$ 2 mô hình tổ hợp};

    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{itemize}
        \item Bỏ cổng: mô hình không suy ra được \bad{loại tấn công từ số
              cổng}.
        \item Bỏ trùng \emph{trước} khi chia: không luồng nào ở cả hai tập.
        \item Xếp hạng đặc trưng chỉ tính trên \emph{tập huấn luyện}.
        \item Chọn mô hình trên \emph{tập validation}. Tập kiểm tra
              \bad{chỉ dùng một lần}, ở cuối.
      \end{itemize}
    \end{column}
  \end{columns}
  \note{\say Đây là quy trình, đi từ trên xuống. Hai khối đỏ là chỗ chặn rò
  rỉ. Trước khi chia dữ liệu, chúng tôi bỏ cổng và bỏ luồng trùng, nên không
  luồng nào nằm ở cả hai tập. Sau đó áp dụng bốn cách giảm đặc trưng, huấn
  luyện tám thuật toán. Bảng xếp hạng đặc trưng chỉ tính trên tập huấn luyện.
  Chọn mô hình thì dùng tập validation. Tập kiểm tra chỉ dùng một lần, ở cuối.
  Cuối cùng, cấu hình SHAP 30 đặc trưng được đưa xuống thiết bị biên.
  \ifasked Tập validation tách 20\% từ tập huấn luyện. Bốn cấu hình so sánh
  được cố định từ trước (giữ hết, RF Top-30, SHAP Top-30, PCA $k{=}40$), siêu
  tham số dùng chung, nên khác biệt phản ánh phương pháp chứ không phải việc
  tinh chỉnh.}
\end{frame}

% ---------------------------------------------------------------- 0:50
\begin{frame}{Bỏ luồng trùng, nhưng phải bỏ cho đúng}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{0.4em}
        \item Nhiều luồng \key{giống hệt nhau} ở mọi đặc trưng
              $\rightarrow$ chỉ giữ \emph{một} bản.
        \item Có nhóm giống hệt nhau mà \bad{nhãn lại khác nhau} (vừa Benign
              vừa Attack), một lỗi đã biết của CICIDS2017.
        \item Nhóm như vậy \bad{bỏ cả nhóm}: \num{270 nhóm, 44.260 dòng}.
        \item Nếu giữ ngẫu nhiên một dòng, \emph{mỗi lần chạy ra một bộ dữ
              liệu khác}. Bỏ cả nhóm thì kết quả luôn như nhau.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering\footnotesize
      \fitwidth{%
      \begin{tabular}{lr}
        \toprule
        \multicolumn{2}{l}{\textbf{Dữ liệu sau khi làm sạch}} \\
        \midrule
        Đặc trưng                     & 77 \\
        Luồng huấn luyện              & 1.785.349 \\
        Luồng kiểm tra                & 447.275 \\
        \midrule
        CSE-CIC-IDS2018 (lấy mẫu 20\%) & $\approx$2,39 triệu \\
        \bottomrule
      \end{tabular}
      }
      \\[0.6em]
      \raggedright\small
      Huấn luyện trên cụm Spark gồm \key{2 Jetson Orin Nano Super}, máy chủ
      chỉ điều phối.
    \end{column}
  \end{columns}
  \note{\say Bỏ luồng trùng nghe đơn giản nhưng dễ làm sai. Những luồng giống
  hệt nhau ở mọi đặc trưng thì chỉ giữ một bản. Nhưng có những nhóm giống hệt
  nhau mà nhãn lại khác nhau, vừa bình thường vừa tấn công. Đây là lỗi đã biết
  của CICIDS2017. Với nhóm như vậy chúng tôi bỏ cả nhóm: 270 nhóm, hơn 44 nghìn
  dòng. Lý do là nếu giữ ngẫu nhiên một dòng thì mỗi lần chạy ra một bộ dữ
  liệu khác. Sau khi làm sạch còn 77 đặc trưng, khoảng 1,8 triệu luồng huấn
  luyện và 450 nghìn luồng kiểm tra.
  \ifasked Cột bị bỏ: destination\_port, source\_port, protocol và các cột định
  danh. Lỗi nhãn mâu thuẫn đã được Lanvin et al.\ ghi nhận. CSE-CIC-IDS2018 lấy
  mẫu phân tầng 20\%, được 3,14 triệu luồng, còn 2,39 triệu sau khi bỏ trùng.
  Spark 3.4.1. Máy chủ chỉ làm master, không chạy executor, nên thời gian huấn
  luyện là thời gian trên hai Jetson.}
\end{frame}

\begin{frame}{Kết quả 1: bỏ 60\% đặc trưng, F1 gần như không đổi}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \centering
      \fitwidth{%
      \begin{tabular}{lccc}
        \toprule
        Cách giảm & Đặc trưng & F1 & Phút huấn luyện \\
        \midrule
        Giữ hết              & 77          & 0,9971       & 77 \\
        \textbf{SHAP Top-30} & \num{30}    & \num{0,9970} & \num{20} \\
        PCA $k{=}40$         & 40          & 0,9939       & 24 \\
        RF Top-30            & 30          & 0,9922       & 22 \\
        \bottomrule
      \end{tabular}
      }
      \\[0.3em]{\scriptsize XGBoost, một lần chạy trên cách chia train/test gốc.}
    \end{column}
    \begin{column}{0.37\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{0.4em}
        \item SHAP giữ \good{30/77 đặc trưng}, F1 gần như không đổi, huấn
              luyện \good{nhanh gấp 3,8 lần}.
        \item Cùng 30 đặc trưng, \key{SHAP chọn tốt hơn RF}.
        \item PCA cần 40 thành phần và \bad{không giải thích được}.
      \end{itemize}
    \end{column}
  \end{columns}
  \takeaway{Chọn \key{SHAP Top-30}: ít đặc trưng, huấn luyện nhanh, giải thích
  được.}
  \note{\say Kết quả đầu tiên. Bảng này là XGBoost với bốn cách giảm đặc
  trưng. Giữ hết 77 đặc trưng thì F1 là 0,9971. SHAP chỉ giữ 30 đặc trưng mà F1
  vẫn 0,9970, và thời gian huấn luyện giảm từ 77 phút xuống 20 phút, tức nhanh
  gấp 3,8 lần. Cùng chọn 30 đặc trưng thì SHAP cho F1 cao hơn RF, 0,9970 so với
  0,9922. PCA thì cần 40 thành phần, điểm thấp hơn, và không giải thích được.
  Vậy SHAP Top-30 là cấu hình chúng tôi chọn.
  \ifasked SHAP hơn RF ở cả bốn thuật toán cây, xem slide dự phòng. Thời gian
  huấn luyện chính xác là 4615,5 s và 1226,1 s, giảm khoảng 73\%. PCA dùng
  $k{=}40$ vì nó trích xuất chứ không chọn lọc: mỗi thành phần chỉ mang một
  phần phương sai.}
\end{frame}

\begin{frame}{Kết quả 2: cổng đích làm tăng điểm, mức độ tùy mô hình}

    \end{column}
    \begin{column}{0.37\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{0.4em}
        \item \key{Mô hình cây} hầu như không cần cổng.
        \item \bad{Mô hình tuyến tính} dựa vào cổng nhiều, \bad{gấp hơn 20
              lần} mô hình cây.
      \end{itemize}
    \end{column}
  \end{columns}
  \takeaway{Đo rò rỉ trên \emph{một} mô hình không nói được gì cho mô hình khác.

  \note{\say Kết quả thứ hai: cổng đích làm tăng điểm đến mức nào. Mỗi thuật
  toán được huấn luyện hai lần, một lần bỏ cổng và một lần giữ cổng. Với bốn mô
  hình cây, giữ cổng chỉ tăng F1 chưa tới 0,002. Với hai mô hình tuyến tính thì
  tăng khoảng 0,04, tức gấp hơn 20 lần. Bài học là đo rò rỉ trên một mô hình
  thì không nói được gì cho mô hình khác. Từ đây trở đi, mọi kết quả đều đã bỏ
  cổng.
  \ifasked Số từng thuật toán ở slide dự phòng. Mô hình cây dựng lại được cùng
  ranh giới từ thống kê hành vi của luồng. Mô hình tuyến tính không làm được,
  nên dựa vào cổng như một định danh dịch vụ. Bản nộp ban đầu chỉ đo trên Random
  Forest (0,9964 so với 0,9954). Bản camera-ready đo cả tám thuật toán theo góp
  ý phản biện.}
\end{frame}

\begin{frame}{Dự phòng: các yếu tố ảnh hưởng tính hợp lệ}
  \begin{itemize}\footnotesize
    \item \textbf{Nội tại.} Xếp hạng RF/SHAP lấy từ tập huấn luyện gốc và giữ
          cố định qua các lần lấy mẫu lại, nhất quán với thiết kế cố định từ
          trước, nhưng có thể gây rò rỉ nhẹ ở phần xếp hạng. Xếp hạng lại trong
          hai trong sáu lần lấy mẫu chỉ làm F1 lệch $\le 7{,}4\times10^{-5}$,
          và 25--26/30 đặc trưng vẫn được chọn lại. MLP mang hạn chế không có
          trọng số lớp.
    \item \textbf{Ngoại tại.} CICIDS2017 có thể không phản ánh lưu lượng hiện
          tại, và các bộ dữ liệu NIDS chuẩn đều có hạn chế chất lượng đã ghi
          nhận. Kết luận liên bộ dữ liệu dựa trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018, chưa phải toàn bộ. Chưa kiểm thử lưu lượng né tránh.
          Bản thân việc khử trùng lặp cũng làm thay đổi phân bố benign/attack.
    \item \textbf{Thống kê.} $N{=}6$ cho kiểm định hoán vị \emph{power} thấp:
          liệt kê đầy đủ hai phía chặn $p$ ở mức $2/2^N$, nên $N{=}6$ \emph{vừa
          đủ} chạm 0,05. Kết luận tương đương không dựa vào $p$ đó mà dựa vào
          TOST, trên cùng sai số Nadeau--Bengio đã tính đến việc các lần lấy
          mẫu chồng nhau. F1 liên bộ không tất định giữa các lượt, nên được báo
          trên 18 lượt. Kiểm định chéo lồng với $N$ lớn hơn là hướng củng cố
          tiếp theo.
  \end{itemize}
\end{frame}
