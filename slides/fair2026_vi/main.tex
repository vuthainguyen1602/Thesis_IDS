\begin{frame}{F1 trên 0,99 với CICIDS2017: điểm đó có đáng tin không?}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \textbf{Hai câu hỏi}
      \begin{enumerate}
        \item \key{Cần cả $\approx$80 đặc trưng không?}\\
              {\small Nhiều đặc trưng thì tốn tính toán, khó chạy trên
              thiết bị biên.}
        \item \key{Điểm cao có đáng tin không?}\\
              {\small Hai dạng \key{rò rỉ dữ liệu} làm điểm cao hơn thực tế:}
              \begin{itemize}\small
                \item Cổng đích: mỗi loại tấn công nhắm vào một số cổng cố định,
                      nên \bad{cổng gần như tiết lộ nhãn}.
                \item Luồng trùng lặp ở \emph{cả hai} tập: mô hình \bad{đã
                      thấy dữ liệu kiểm tra khi huấn luyện}.
              \end{itemize}
      \end{enumerate}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{alertblock}{Bài báo này}
        Sửa quy trình đánh giá \emph{trước}, rồi mới so sánh các phương
        pháp giảm đặc trưng.
      \end{alertblock}
      \begin{exampleblock}{Thông điệp}
        \key{Đánh giá đúng cách} quan trọng không kém \emph{chọn đúng phương pháp}.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{\say Bài báo xuất phát từ hai câu hỏi. Thứ nhất, mỗi luồng mạng có
  khoảng 80 đặc trưng, có cần dùng hết không, vì càng nhiều thì càng tốn và
  càng khó chạy trên thiết bị biên. Thứ hai, rất nhiều bài đạt F1 trên 0,99 với
  bộ dữ liệu này, điểm đó có đáng tin không. Có hai dạng rò rỉ dữ liệu khiến
  điểm cao hơn thực tế. Rò rỉ dữ liệu, nói đơn giản, là mô hình được biết
  trước thứ mà khi triển khai thật nó không có. Một
  là cổng đích: trong bộ dữ liệu, mỗi loại tấn công nhắm vào một số cổng cố định, nên cổng gần như tiết lộ nhãn. Hai là luồng trùng nhau nằm ở cả tập huấn luyện
  lẫn tập kiểm tra, mô hình xem như đã thấy dữ liệu kiểm tra từ lúc huấn luyện. Vì vậy chúng
  tôi sửa quy trình đánh giá trước, rồi mới so sánh các phương pháp giảm đặc trưng.
  \ifasked Rò rỉ dữ liệu là khi thông tin không có lúc triển khai lọt vào huấn
  luyện hoặc đánh giá. Cột cổng đích trong dữ liệu là
  \texttt{destination\_port}.}
\end{frame}

\begin{frame}{Các đóng góp}
  \begin{enumerate}
    \setlength{\itemsep}{0.35em}
    \item So sánh \key{4 phương pháp giảm đặc trưng} (giữ toàn bộ, RF
          Importance, SHAP, PCA) trên \emph{cùng} 8 thuật toán Spark ML.
    \item \key{Quy trình đánh giá chống rò rỉ}: bỏ cổng đích, bỏ luồng trùng
          lặp trước khi chia dữ liệu, tập kiểm tra chỉ dùng một lần ở cuối.
    \item \key{Kiểm định thống kê} để kết luận, thay vì chỉ so sánh điểm số.
    \item Đánh giá hai khía cạnh mà điểm F1 tổng che khuất: \key{từng loại
          tấn công} và \key{đổi sang bộ dữ liệu khác}.
    \item Chọn cấu hình \key{phù hợp thiết bị biên} Jetson Orin Nano Super.
  \end{enumerate}
  \note{\say Bài báo có năm đóng góp. Một, so sánh bốn phương pháp giảm đặc trưng
  trên cùng tám thuật toán, cùng một quy trình. Hai, quy trình đánh giá chống
  rò rỉ. Ba, dùng kiểm định thống kê để kết luận, không chỉ so sánh điểm số. Bốn,
  đánh giá những gì điểm tổng thường che khuất: từng loại tấn công, và khi đổi sang bộ
  dữ liệu khác. Năm, chọn cấu hình phù hợp thiết bị biên. Điểm mới là mục
  hai đến bốn ít khi được làm cùng nhau trong các bài về CICIDS2017.}
\end{frame}

  luyện. Chọn mô hình thì dùng tập kiểm định, tức tập validation tách từ tập

  Siêu tham số được cố định từ trước và dùng chung cho cả bốn phương pháp, nên khác
  biệt là do phương pháp chứ không do tinh chỉnh.
  Cuối cùng, cấu hình SHAP 30 đặc trưng được đưa xuống thiết bị biên.
  \ifasked Tập kiểm định tách 20\% từ tập huấn luyện. Bốn cấu hình so sánh
  được cố định từ trước (toàn bộ 77, RF Top-30, SHAP Top-30, PCA $k{=}40$), siêu
  tham số dùng chung, nên khác biệt phản ánh phương pháp chứ không phải việc
  tinh chỉnh.}

\begin{frame}{Kết quả 1: bỏ 60\% đặc trưng, F1 gần như không đổi}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \centering\footnotesize
      \fitwidth{%
      \begin{tabular}{lccc}
        \toprule
        Phương pháp & Đặc trưng & F1 & Phút \\
        \midrule
        Toàn bộ              & 77          & 0,9971       & 77 \\
        \textbf{SHAP Top-30} & \num{30}    & \num{0,9970} & \num{20} \\
        PCA $k{=}40$         & 40          & 0,9939       & 24 \\
        RF Top-30            & 30          & 0,9922       & 22 \\
        \bottomrule
      \end{tabular}
      }
      \\[0.3em]{\scriptsize XGBoost, một lần chạy. Phút = thời gian huấn luyện.}
    \end{column}
    \begin{column}{0.45\textwidth}
      \footnotesize
      \stat{accGreen}{30/77}{đặc trưng giữ lại}
        {F1 0,9971 $\to$ 0,9970, gần như không đổi.}
      \stat{accGreen}{3,8$\times$}{huấn luyện nhanh hơn}
        {77 phút $\to$ 20 phút.}
      \stat{accBlue}{SHAP $>$ RF}{khi cùng chọn 30}
        {0,9970 so với 0,9922. PCA 0,9939, \bad{không giải thích được}.}
    \end{column}
  \end{columns}
  \takeaway{Chọn \key{SHAP Top-30}: ít đặc trưng, huấn luyện nhanh, giải thích
  được.}
  \note{\say Kết quả đầu tiên, trên XGBoost. Ba câu hỏi. Một, bỏ 47 trên 77
  đặc trưng thì F1 có giảm không? Gần như không: 0,9971 xuống 0,9970. Hai,

  sang mạng khác, mô hình bỏ sót gần hết tấn công. Bốn, 1\%: chỉ cần 1\% dữ
  liệu có nhãn của mạng mới, khoảng 19 nghìn luồng, để huấn luyện lại. Bài học chung: đánh giá đúng cách
  quan trọng không kém chọn đúng phương pháp. Xin cảm ơn quý thầy cô, kính mời
  câu hỏi.
  \ifasked Hướng tiếp theo: đo riêng tác động của bước bỏ trùng lặp; toàn bộ
  CSE-CIC-IDS2018 và các bộ mới hơn (RoEduNet, CIC-IoT); thử bộ 30 đặc trưng
  SHAP trên bộ dữ liệu khác. Tăng số lần lặp bằng kiểm định
  chéo lồng cũng là hướng củng cố.}
\end{frame}

\begin{frame}{Dự phòng: các yếu tố ảnh hưởng tính hợp lệ}
  \begin{itemize}\footnotesize
    \item \textbf{Nội tại.} Xếp hạng RF/SHAP lấy từ tập huấn luyện gốc và giữ
          cố định qua các lần lấy mẫu lại, nhất quán với thiết kế cố định từ
          trước, nhưng có thể gây rò rỉ nhẹ ở phần xếp hạng. Xếp hạng lại trong
          hai trong sáu lần lấy mẫu chỉ làm F1 lệch $\le 7{,}4\times10^{-5}$,
          và 25--26/30 đặc trưng vẫn được chọn lại. MLP mang hạn chế không có
          trọng số lớp.
    \item \textbf{Ngoại tại.} CICIDS2017 có thể không phản ánh lưu lượng hiện
          tại, và các bộ dữ liệu NIDS chuẩn đều có hạn chế chất lượng đã ghi
          nhận. Kết luận liên bộ dữ liệu dựa trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018, chưa phải toàn bộ. Chưa kiểm thử lưu lượng né tránh.
          Bản thân việc khử trùng lặp cũng làm thay đổi phân bố benign/attack.
    \item \textbf{Thống kê.} $N{=}6$ cho kiểm định hoán vị \emph{power} thấp:
          liệt kê đầy đủ hai phía chặn $p$ ở mức $2/2^N$, nên $N{=}6$ \emph{vừa
          đủ} chạm 0,05. Kết luận tương đương không dựa vào $p$ đó mà dựa vào
          TOST, trên cùng sai số Nadeau--Bengio đã tính đến việc các lần lấy
          mẫu chồng nhau. F1 liên bộ dữ liệu không tất định giữa các lượt chạy, nên được báo
          trên 18 lượt. Kiểm định chéo lồng với $N$ lớn hơn là hướng củng cố
          tiếp theo.
  \end{itemize}
\end{frame}
