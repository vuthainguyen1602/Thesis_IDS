\begin{frame}{F1 gần như tuyệt đối trên CICIDS2017: nên nghi ngờ}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Hai câu hỏi}
      \begin{enumerate}
        \item \key{Có cần đủ $\approx$80 đặc trưng không?}\\
              {\small Nhiều đặc trưng thì tốn tính toán, khó chạy trên thiết
              bị biên.}
        \item \key{Điểm số cao có đáng tin không?}\\
              {\small Hai lỗi phổ biến làm điểm cao giả:}
              \begin{itemize}\small
                \item \texttt{destination\_port} \bad{lộ luôn loại tấn công}.
                \item Luồng trùng nhau nằm ở \emph{cả} tập huấn luyện
                      \emph{lẫn} tập kiểm tra: mô hình \bad{được xem trước
                      đáp án}.
              \end{itemize}
      \end{enumerate}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{alertblock}{Bài báo này}
        Sửa cách đánh giá cho đúng \emph{trước}, rồi mới so sánh các cách
        giảm đặc trưng.
      \end{alertblock}
      \begin{exampleblock}{Thông điệp}
        \key{Đo thế nào} quan trọng ngang với \emph{phương pháp nào thắng}.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{Rò rỉ dữ liệu là khi thông tin không có lúc triển khai lọt vào huấn
  luyện hoặc đánh giá. CICIDS2017 bắn mỗi loại tấn công vào những cổng cố định,
  nên cổng đích gần như là nhãn. Cả hai lỗi đều được sửa trước khi chia tập.}
\end{frame}

% ---------------------------------------------------------------- 0:35
\begin{frame}{Các đóng góp}
  \begin{enumerate}
    \setlength{\itemsep}{0.35em}
    \item So sánh \key{4 cách giảm đặc trưng} (giữ hết, RF Importance, SHAP,
          PCA) trên \emph{cùng} 8 thuật toán Spark ML.
    \item \key{Quy trình đánh giá chống rò rỉ}: bỏ cổng, bỏ luồng trùng trước
          khi chia, không dùng tập kiểm tra để chọn bất cứ thứ gì.

    \item Soi những chỗ điểm tổng hay che giấu: \key{từng loại tấn công}, và
          khi \key{đổi sang bộ dữ liệu khác}.
    \item Chọn cấu hình \key{hợp với thiết bị biên} Jetson Orin Nano Super.
  \end{enumerate}
  \note{Mục 2 đến 4 là những thứ ít khi được làm cùng nhau trong các bài về
  CICIDS2017.}
\end{frame}

\begin{frame}{Kết quả 3: giữ hết hay giữ 30 đặc trưng --- thực chất như nhau}

      \begin{itemize}
        \setlength{\itemsep}{0.4em}
        \item Lặp lại với \key{6 cách chia dữ liệu khác nhau}, so từng cặp.
        \item Không chỉ hỏi ``có khác không'' mà \key{chứng minh} hai bên
              tương đương (kiểm định TOST).
        \item Kết quả: chênh lệch thật \good{nhỏ hơn 0,00023 F1}.
      \end{itemize}
    \end{column}
  \end{columns}
  \takeaway{Hai cấu hình tương đương, nên chọn \key{SHAP Top-30}: ít đặc trưng
  hơn và giải thích được.}
  \note{Nếu bị hỏi chi tiết: kiểm định hoán vị ghép cặp liệt kê đủ $2^6$ tổ hợp
  dấu cho $p = 0{,}031$, đúng sàn phân giải $2/2^6$, Cohen's $d \approx 1{,}94$.
  Một $p$ như vậy không chứng minh được tương đương, nên dùng TOST với biên
  $\pm0{,}001$ F1 khai báo trước, trên sai số hiệu chỉnh Nadeau--Bengio:
  $p_{\text{TOST}} = 3{,}9\times10^{-6}$, biên nhỏ nhất chứng minh được là
  $\pm2{,}3\times10^{-4}$. Xem slide dự phòng.}
\end{frame}

\begin{frame}{Dự phòng: các yếu tố ảnh hưởng tính hợp lệ}
  \begin{itemize}\footnotesize
    \item \textbf{Nội tại.} Xếp hạng RF/SHAP lấy từ tập huấn luyện gốc và giữ
          cố định qua các lần lấy mẫu lại --- nhất quán với thiết kế cố định từ
          trước, nhưng có thể gây rò rỉ nhẹ ở phần xếp hạng. Xếp hạng lại trong
          hai trong sáu lần lấy mẫu chỉ làm F1 lệch $\le 7{,}4\times10^{-5}$,
          và 25--26/30 đặc trưng vẫn được chọn lại. MLP mang hạn chế không có
          trọng số lớp.
    \item \textbf{Ngoại tại.} CICIDS2017 có thể không phản ánh lưu lượng hiện
          tại, và các bộ dữ liệu NIDS chuẩn đều có hạn chế chất lượng đã ghi
          nhận. Kết luận liên bộ dữ liệu dựa trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018, chưa phải toàn bộ. Chưa kiểm thử lưu lượng né tránh.
          Bản thân việc khử trùng lặp cũng làm thay đổi phân bố benign/attack.
    \item \textbf{Thống kê.} $N{=}6$ cho kiểm định hoán vị \emph{power} thấp:
          liệt kê đầy đủ hai phía chặn $p$ ở mức $2/2^N$, nên $N{=}6$ \emph{vừa
          đủ} chạm 0,05. Kết luận tương đương không dựa vào $p$ đó mà dựa vào
          TOST, trên cùng sai số Nadeau--Bengio đã tính đến việc các lần lấy
          mẫu chồng nhau. F1 liên bộ không tất định giữa các lượt, nên được báo
          trên 18 lượt. Kiểm định chéo lồng với $N$ lớn hơn là hướng củng cố
          tiếp theo.
  \end{itemize}
\end{frame}
