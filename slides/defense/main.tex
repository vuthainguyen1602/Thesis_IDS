\section{I.\ ĐẶT VẤN ĐỀ VÀ MỤC TIÊU NGHIÊN CỨU}
% =====================================================================

\begin{frame}{Bối cảnh và ba khoảng trống nghiên cứu}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Vấn đề thực tiễn}
      \begin{itemize}\footnotesize
        \item Thiết bị IoT tăng nhanh, tài nguyên ít, bảo mật yếu.
        \item Lưu lượng lớn, liên tục $\rightarrow$ cần nền tảng \key{phân tán}.
        \item Phát hiện cần đặt \key{gần nguồn dữ liệu} để giảm băng thông, độ trễ.
      \end{itemize}
      \vspace{0.3em}
      \textbf{Ba khoảng trống}
      \begin{enumerate}\footnotesize
        \item RF, SHAP, PCA hiếm khi so sánh trên \emph{cùng một} pipeline phân tán.
        \item Điểm số trên CICIDS2017 thường \bad{gần như hoàn hảo} --- dấu hiệu
              rò rỉ dữ liệu.
        \item IDS biên thường chỉ đo trên \bad{một thiết bị}.
      \end{enumerate}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{alertblock}{Câu hỏi trung tâm}
        \footnotesize Làm sao xây dựng pipeline IDS trên Spark vừa \emph{đánh giá
        trung thực} ở pha offline, vừa \emph{chạy thời gian thực} trên cụm biên?
      \end{alertblock}
      \begin{exampleblock}{Cách tiếp cận: hai pha bổ trợ}
        \footnotesize \key{(A) Offline} --- mô hình nào, bao nhiêu đặc trưng,
        dưới quy trình loại trừ rò rỉ.\\[0.2em]
        \key{(B) Biên} --- mô hình đó chạy được thời gian thực không?
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{Nhấn mạnh: hai pha trả lời hai câu hỏi khác nhau, không phải hai
  phương án cạnh tranh.}
\end{frame}

\begin{frame}{Mục tiêu và ba câu hỏi nghiên cứu}
  \textbf{Mục tiêu.} Xây dựng hệ thống IDS kết hợp học máy với Apache Spark,
  đánh giá dưới quy trình \key{loại trừ rò rỉ dữ liệu} có kiểm định thống kê, và
  kiểm chứng khả năng chạy thời gian thực trên cụm thiết bị biên.
  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ1 --- Giảm chiều}
        \footnotesize Giảm chiều/chọn đặc trưng có giữ được hiệu năng gần mức
        dùng toàn bộ đặc trưng không, khi cắt hơn 60\% số đặc trưng?
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ2 --- Học tổ hợp}
        \footnotesize Học tổ hợp có lợi thế \emph{đáng kể} so với mô hình đơn
        tốt nhất không, khi được \emph{kiểm định thống kê}?
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ3 --- Khả thi ở biên}
        \footnotesize Pipeline Spark có khả thi về thông lượng, độ trễ, năng
        lượng khi triển khai phân tán trên cụm Jetson không?
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \footnotesize
  \textbf{Phạm vi.} CICIDS2017 (mở rộng CSE-CIC-IDS2018 cho thí nghiệm liên bộ),
  nhị phân Benign/Attack, cụm 2 Jetson Orin Nano Super. Bộ dữ liệu \emph{không} có
  lưu lượng IoT đặc thù --- tính chất ``IoT'' nằm ở \key{kịch bản triển khai} trên
  thiết bị biên; tấn công đối kháng \bad{ngoài phạm vi}.
  \note{Đọc chậm ba RQ; hội đồng sẽ đối chiếu với phần kết luận. Nêu chủ động giới
  hạn phạm vi IoT --- slide dự phòng có bản đầy đủ kèm mô hình mối đe dọa.}
\end{frame}

\begin{frame}{Pha A --- Quy trình loại trừ rò rỉ và bốn chiến lược so sánh}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Hai nguồn rò rỉ, xử lý \emph{trước khi chia tập}}\vspace{-0.2em}
      \begin{enumerate}\footnotesize
        \item \key{Đặc trưng rò rỉ nhãn}: \texttt{destination\_port} mã hóa trực
              tiếp dịch vụ bị tấn công $\rightarrow$ mô hình học thuộc ánh xạ
              cổng\,$\rightarrow$\,nhãn thay vì học hành vi luồng.
        \item \key{Trùng lặp ở mức đặc trưng}: luồng giống hệt nhau trên
              \emph{toàn bộ} cột đặc trưng không bị loại bởi khử trùng lặp toàn
              dòng $\rightarrow$ cùng một mẫu vào cả hai tập.
      \end{enumerate}
      \vspace{0.2em}
      \footnotesize Nhóm \emph{nhãn xung đột} bị \bad{loại cả nhóm} (270 nhóm,
      44.260 dòng); còn 77 đặc trưng, 1,79\,tr / 0,45\,tr.
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Bốn chiến lược, cùng một pipeline}
      \begin{itemize}\footnotesize
        \item \textbf{Baseline} 77 đặc trưng $\cdot$ \textbf{RF Top-$K$} $\cdot$
              \textbf{SHAP Top-$K$} $\cdot$ \textbf{PCA $k$=40}; xếp hạng RF/SHAP
              chỉ dùng \emph{tập huấn luyện}.
        \item \textbf{8 thuật toán}: DT, LR, SVM, NB, RF, GBT, XGBoost, MLP.
        \item \textbf{2 mô hình học tổ hợp}: \key{Soft Voting} và \key{Hybrid
              Bagging} --- Top-3 theo F1 trên tập kiểm định.
      \end{itemize}
      \vspace{0.2em}
      \footnotesize \textbf{Chống thiên lệch}: cấu hình cố định trước, chọn mô hình
      trên \key{tập kiểm định}; tập kiểm tra \bad{không tham gia} khâu lựa chọn.
    \end{column}
  \end{columns}
  \note{Đóng góp phương pháp luận quan trọng nhất. Nói rõ vì sao khử trùng lặp tất
  định lại quan trọng: kết quả không phụ thuộc cách Spark phân vùng dữ liệu. PCA
  dùng k=40 vì là phép trích xuất --- slide dự phòng có lý do đầy đủ.}
\end{frame}

      \fitwidth{%
      \begin{tabular}{llccc}

        \midrule

        XGBoost       & 0,9922 & 0,9970 & \good{$+$0,48\%} \\
        GBT           & 0,9844 & 0,9918 & \good{$+$0,75\%} \\
        Decision Tree & 0,9835 & 0,9948 & \good{$+$1,15\%} \\
        \bottomrule
      \end{tabular}

    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{itemize}\small
        \item SHAP Top-30 giữ F1 ngang baseline, cắt \good{$\approx$60\% đặc
              trưng} và \good{$\approx$73\% thời gian huấn luyện}.
        \item Ở \emph{cùng} số đặc trưng, SHAP vượt RF Importance trên mọi
              thuật toán họ cây.
        \item Học tổ hợp \key{Hybrid Bagging} với SHAP Top-30 đạt F1
              \num{0,9966} --- cao nhất trong các cấu hình đã thử.
        \item Ít đặc trưng hơn $\rightarrow$ chi phí \texttt{VectorAssembler}
              và bộ nhớ mô hình ở biên nhỏ hơn.
      \end{itemize}
    \end{column}
  \end{columns}

    \begin{column}{0.50\textwidth}
      \begin{itemize}\small
        \item $N{=}6$ lần \emph{lấy mẫu lại} ghép cặp $\rightarrow$ phương sai
              phản ánh tác động của lấy mẫu dữ liệu, không chỉ của seed.
        \item Kiểm định hoán vị hai phía \key{liệt kê đầy đủ} $2^6$ hoán vị dấu:
              $p$ nhỏ nhất là $2/2^6$, nên \key{bắt buộc $N \geq 6$}.
        \item Các lần lấy mẫu \emph{chồng lấn} nên vi phạm giả định độc lập của
              phân phối $t$ $\rightarrow$ kết luận theo khoảng tin cậy hiệu chỉnh

  \end{columns}

  định, chênh lệch tuyệt đối $\approx$0,0001 $\rightarrow$ hai cấu hình

\begin{frame}[label=bk-pca]{Dự phòng --- Vì sao PCA dùng $k$=40, và vì sao loại LightGBM?}
  \begin{block}{PCA $k$=40 thay vì $k$=30}
    \small
    PCA là phép \emph{trích xuất}, không phải \emph{chọn lọc}: mỗi thành phần
    chính là một tổ hợp tuyến tính chỉ mang một phần phương sai, nên cần nhiều
    thành phần hơn để giữ lượng thông tin tương đương tập 30 đặc trưng gốc.
    Trong dải quét chung $\{20,30,40\}$, $k$=40 là mức giữ phương sai cao nhất,
    nên là điểm vận hành đại diện \emph{công bằng nhất} cho PCA; ép $k$=30 sẽ
    làm PCA mất thêm phương sai và bất lợi khi so sánh.
  \end{block}
  \begin{block}{LightGBM bị loại khỏi so sánh}
    \small
    Thư viện native của LightGBM (qua SynapseML) chỉ hỗ trợ kiến trúc x86\_64,
    không chạy được trên thiết bị biên Jetson Orin Nano Super (ARM64) --- vốn là
    \emph{đích triển khai} của luận văn. Trong họ gradient boosting, XGBoost và
    GBT đóng vai trò đại diện.
  \end{block}
\end{frame}

\begin{frame}[label=bk-threats]{Dự phòng --- Đe dọa đến tính hợp lệ}
  \begin{itemize}\small
    \item \textbf{Nội tại.} Xếp hạng RF/SHAP lấy từ tập huấn luyện gốc và giữ
          cố định qua các lần lấy mẫu lại --- nhất quán với thiết kế ``cấu hình
          cố định trước'', nhưng có thể gây rò rỉ nhẹ ở phần xếp hạng. Xếp hạng
          lại theo từng lần lấy mẫu chặt chẽ hơn nhưng rất tốn kém với SHAP.
          MLP không được gán trọng số lớp (Spark ML không hỗ trợ).
    \item \textbf{Ngoại tại.} CICIDS2017 (2017) có thể không phản ánh lưu lượng
          hiện tại; các bộ dữ liệu NIDS chuẩn đều có hạn chế chất lượng đã ghi
          nhận. Kết luận liên bộ dữ liệu dựa trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018. Chưa đánh giá trước lưu lượng đối kháng. Bản thân
          việc khử trùng lặp cũng làm thay đổi phân bố benign/attack.
    \item \textbf{Thống kê.} $N{=}6$ cho \emph{power} thấp: kiểm định hai phía
          liệt kê đầy đủ chặn $p$ ở mức $2/2^N$, nên $N{=}6$ chỉ vừa đủ chạm
          ngưỡng 0,05. Các lần lấy mẫu chồng lấn, vi phạm giả định độc lập của
          phân phối $t$ --- do đó Nadeau--Bengio là căn cứ chính. Tăng $N$ với
          cross-validation lồng nhau là hướng củng cố tiếp theo.
  \end{itemize}
\end{frame}

    \item Mọi truy vấn chỉ số canh đúng cửa sổ tải, không dùng khoảng thời gian

    \item Phân vị độ trễ tính \emph{theo từng nút} trên mẫu thô từng luồng; p95
