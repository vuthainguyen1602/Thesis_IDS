\section{I.\ ĐẶT VẤN ĐỀ VÀ MỤC TIÊU NGHIÊN CỨU}
% =====================================================================

\begin{frame}{Bối cảnh và ba khoảng trống nghiên cứu}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Vấn đề thực tiễn}
      \begin{itemize}\footnotesize
        \item Thiết bị IoT tăng nhanh, tài nguyên ít, bảo mật yếu.
        \item Lưu lượng lớn, liên tục $\rightarrow$ cần nền tảng \key{phân tán}.
        \item Phát hiện cần đặt \key{gần nguồn dữ liệu} để giảm băng thông, độ trễ.
      \end{itemize}
      \vspace{0.3em}
      \textbf{Ba khoảng trống}
      \begin{enumerate}\footnotesize
        \item RF, SHAP, PCA hiếm khi so sánh trên \emph{cùng một} pipeline phân tán.
        \item Điểm số trên CICIDS2017 thường \bad{gần như hoàn hảo} --- dấu hiệu
              rò rỉ dữ liệu.
        \item IDS biên thường chỉ đo trên \bad{một thiết bị}.
      \end{enumerate}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{alertblock}{Câu hỏi trung tâm}
        \footnotesize Làm sao xây dựng pipeline IDS trên Spark vừa \emph{đánh giá
        trung thực} ở pha offline, vừa \emph{chạy thời gian thực} trên cụm biên?
      \end{alertblock}
      \begin{exampleblock}{Cách tiếp cận: hai pha bổ trợ}
        \footnotesize \key{(A) Offline} --- mô hình nào, bao nhiêu đặc trưng,
        dưới quy trình loại trừ rò rỉ.\\[0.2em]
        \key{(B) Biên} --- mô hình đó chạy được thời gian thực không?
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{Nhấn mạnh: hai pha trả lời hai câu hỏi khác nhau, không phải hai
  phương án cạnh tranh.}
\end{frame}

\begin{frame}{Mục tiêu và ba câu hỏi nghiên cứu}
  \textbf{Mục tiêu.} Xây dựng hệ thống IDS kết hợp học máy với Apache Spark,
  phát hiện các dạng tấn công phổ biến trên CICIDS2017; thiết kế pipeline xử lý
  phân tán; đánh giá khả năng triển khai trên thiết bị biên; áp dụng quy trình
  đánh giá \key{loại trừ rò rỉ dữ liệu} có kiểm định thống kê.
  \vspace{0.7em}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ1 --- Giảm chiều}
        \footnotesize Giảm chiều/chọn đặc trưng có giữ được hiệu năng gần mức
        dùng toàn bộ đặc trưng không, khi cắt hơn 60\% số đặc trưng?
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ2 --- Học tổ hợp}
        \footnotesize Học tổ hợp có lợi thế \emph{đáng kể} so với mô hình đơn
        tốt nhất không, khi được \emph{kiểm định thống kê}?
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ3 --- Khả thi ở biên}
        \footnotesize Pipeline Spark có khả thi về thông lượng, độ trễ, năng
        lượng khi triển khai phân tán trên cụm Jetson không?
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \centering\footnotesize
  \key{Ba câu hỏi này được trả lời trực tiếp ở slide kết luận.}
  \note{Đọc chậm ba RQ. Hội đồng sẽ đối chiếu chúng với phần kết luận.}
\end{frame}

\begin{frame}{Phạm vi, dữ liệu và mô hình mối đe dọa}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \textbf{Phạm vi}
      \begin{itemize}\small
        \item \textbf{Dữ liệu}: CICIDS2017; mở rộng sang CSE-CIC-IDS2018 cho
              thí nghiệm liên bộ dữ liệu.
        \item \textbf{Công nghệ}: Spark Core/SQL/Streaming/MLlib; Kafka,
              PostgreSQL, InfluxDB, Grafana.
        \item \textbf{Bài toán}: nhị phân Benign/Attack (có bổ sung đánh giá đa
              lớp theo loại tấn công).
        \item \textbf{Triển khai}: cụm 2 Jetson Orin Nano Super (8\,GB) $+$ máy
              chủ điều phối.
      \end{itemize}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{alertblock}{Lưu ý trung thực về phạm vi ``IoT''}
        \footnotesize
        CICIDS2017 là lưu lượng doanh nghiệp \emph{mô phỏng}, không chứa giao
        thức đặc thù IoT (MQTT, CoAP). Tính chất ``IoT'' nằm ở \key{kịch bản
        triển khai}: toàn bộ pipeline suy luận được đo trên \emph{thiết bị biên
        tài nguyên hạn chế} --- lớp gateway trong kiến trúc IoT ba tầng --- chứ
        không ở bản chất lưu lượng huấn luyện.
      \end{alertblock}
      \begin{block}{Mô hình mối đe dọa}
        \footnotesize
        Kẻ tấn công ở bên trong hoặc ở biên mạng; phạm vi giới hạn ở các tấn
        công đã biết kiểu CICIDS2017. Tấn công đối kháng (adversarial/evasion)
        \bad{nằm ngoài phạm vi} đánh giá.
      \end{block}
    \end{column}
  \end{columns}
  \note{Chủ động nêu giới hạn phạm vi IoT trước khi hội đồng hỏi. Đây là câu
  hỏi phản biện gần như chắc chắn sẽ có.}
\end{frame}

\begin{frame}{Pha A --- Quy trình đánh giá loại trừ rò rỉ dữ liệu}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Hai nguồn rò rỉ, xử lý \emph{trước khi chia tập}}
      \begin{enumerate}\small
        \item \key{Đặc trưng rò rỉ nhãn}: \texttt{destination\_port} mã hóa
              trực tiếp dịch vụ bị tấn công $\rightarrow$ mô hình học thuộc ánh
              xạ cổng\,$\rightarrow$\,nhãn thay vì học hành vi luồng.
        \item \key{Trùng lặp ở mức đặc trưng}: luồng giống hệt nhau trên
              \emph{toàn bộ} cột đặc trưng không bị loại bởi phép khử trùng lặp
              toàn dòng $\rightarrow$ \emph{cùng một mẫu} vào cả hai tập.
      \end{enumerate}
      \vspace{0.2em}
      \footnotesize Nhóm luồng giống hệt nhau nhưng \emph{nhãn xung đột} bị
      \bad{loại bỏ cả nhóm}, thay vì giữ lại một dòng tùy cách phân vùng.
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Kết quả tiền xử lý (tất định)}
        \footnotesize
        \fitwidth{%
        \begin{tabular}{lr}
          Nhóm nhãn mâu thuẫn bị loại & 270 nhóm \\
          Số dòng bị loại theo         & 44.260 \\
          Đặc trưng số còn lại         & 77 \\
          Huấn luyện / kiểm tra        & 1,79\,tr / 0,45\,tr \\
        \end{tabular}
        }
      \end{block}
      \begin{exampleblock}{Chống thiên lệch lựa chọn}
        \footnotesize
        \begin{itemize}
          \item Tách \emph{pha thăm dò} khỏi \emph{pha xác nhận}.
          \item Chọn mô hình theo \key{tập kiểm định} tách từ tập huấn luyện.
          \item Tập kiểm tra \bad{không tham gia} bước lựa chọn nào.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{Đây là đóng góp phương pháp luận quan trọng nhất. Nói rõ vì sao khử
  trùng lặp tất định lại quan trọng: kết quả không phụ thuộc cách Spark phân
  vùng dữ liệu.}
\end{frame}

\begin{frame}{Pha A --- Bốn chiến lược giảm chiều và các mô hình so sánh}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Bốn chiến lược, cùng một pipeline Spark}
      \begin{itemize}\small
        \item \textbf{Baseline}: toàn bộ 77 đặc trưng.
        \item \textbf{RF Importance Top-$K$}: xếp hạng nhúng từ Random Forest.
        \item \textbf{SHAP Top-$K$}: giá trị Shapley từ XGBoost, tính trên mẫu
              \key{phân tầng theo lớp}.
        \item \textbf{PCA $k$=40}: trích xuất không giám sát.
      \end{itemize}
      \vspace{0.2em}
      \footnotesize Xếp hạng RF/SHAP chỉ dùng \emph{tập huấn luyện}. Lý do PCA
      dùng $k$=40: xem slide dự phòng.
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{8 thuật toán $+$ 2 mô hình học tổ hợp}
      \begin{itemize}\small
        \item Decision Tree, Logistic Regression, SVM, Naive Bayes, Random
              Forest, GBT, XGBoost, MLP.
        \item \key{Soft Voting} và \key{Hybrid Bagging} --- kết hợp Top-3 thuật
              toán có F1 cao nhất \emph{trên tập kiểm định}.
      \end{itemize}
      \vspace{0.2em}
      \textbf{Mất cân bằng lớp --- xử lý thống nhất}
      \begin{itemize}\small
        \item Trọng số lớp cho 6 mô hình; \texttt{scale\_pos\_weight} cho XGBoost;
              MLP không hỗ trợ.
      \end{itemize}
    \end{column}
  \end{columns}
  \note{Giải thích PCA k=40 chủ động nếu còn thời gian --- đây là câu hỏi phản
  biện hay gặp về tính công bằng của so sánh; slide dự phòng có đầy đủ lý do.
  LightGBM bị loại vì thư viện native chỉ chạy x86\_64, không chạy trên ARM64.}
\end{frame}

\begin{frame}{Pha B --- Ba chế độ triển khai và môi trường đo}

    \begin{column}{0.55\textwidth}
      \begin{description}
        \item[Đơn nút] toàn bộ pipeline trong một tiến trình --- mốc tham chiếu.

          luận văn đã bổ sung đánh giá đa lớp để phân tích lớp hiếm, nhưng một
          bộ phân lớp đa lớp hoàn chỉnh cho thời gian thực vẫn còn ở phía trước.
    \item \textbf{Phạm vi dữ liệu.} CICIDS2017 không chứa lưu lượng IoT đặc thù;
          thí nghiệm liên bộ dữ liệu mới thực hiện trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018, chưa phải toàn bộ.
    \item \textbf{Quy mô.} Khía cạnh xử lý phân tán mới được kiểm chứng ở mức
          \emph{minh chứng khái niệm} trên cụm hai nút 8\,GB với vài triệu
          luồng, chưa phải quy mô terabyte hay cụm nhiều nút mạnh.
    \item \textbf{Đối kháng và tài nguyên chưa khai thác.} Chưa kiểm thử trước
          lưu lượng né tránh (adversarial/evasion); GPU $\approx$67 TOPS còn bỏ
          trống do pipeline hiện chỉ dùng CPU.
  \end{itemize}
  \note{Nêu hạn chế trước khi hội đồng nêu. Mỗi hạn chế có một hướng khắc phục
  tương ứng ở slide sau.}
\end{frame}

\begin{frame}{Hướng phát triển}
  \begin{itemize}\small
    \item \textbf{Cải thiện độ nhạy cho lớp tấn công hiếm.} Xử lý mất cân bằng
          có chủ đích (undersampling/SMOTE áp dụng \emph{sau} khi chia tập),
          focal loss, hạ ngưỡng quyết định theo lớp; \key{kiến trúc phân tầng} ---
          tận dụng chính thiết kế cổng hai tầng: tầng nhị phân ở biên giữ recall
          tổng, còn bộ định danh loại chỉ huấn luyện trên \emph{luồng tấn công},
          huấn luyện lại ngoại tuyến mà không đụng hệ thống đang chạy; thêm

          Jetson; đo độ trễ đầu cuối đầy đủ.

  \end{itemize}
  \note{Mỗi hướng phát triển ứng với một hạn chế ở slide trước --- đó là cách
  thể hiện sự làm chủ đề tài.}
\end{frame}

\begin{frame}{Dự phòng --- Vì sao PCA dùng $k$=40, và vì sao loại LightGBM?}
  \begin{block}{PCA $k$=40 thay vì $k$=30}
    \small
    PCA là phép \emph{trích xuất}, không phải \emph{chọn lọc}: mỗi thành phần
    chính là một tổ hợp tuyến tính chỉ mang một phần phương sai, nên cần nhiều
    thành phần hơn để giữ lượng thông tin tương đương tập 30 đặc trưng gốc.
    Trong dải quét chung $\{20,30,40\}$, $k$=40 là mức giữ phương sai cao nhất,
    nên là điểm vận hành đại diện \emph{công bằng nhất} cho PCA; ép $k$=30 sẽ
    làm PCA mất thêm phương sai và bất lợi khi so sánh.
  \end{block}
  \begin{block}{LightGBM bị loại khỏi so sánh}
    \small
    Thư viện native của LightGBM (qua SynapseML) chỉ hỗ trợ kiến trúc x86\_64,
    không chạy được trên thiết bị biên Jetson Orin Nano Super (ARM64) --- vốn là
    \emph{đích triển khai} của luận văn. Trong họ gradient boosting, XGBoost và
    GBT đóng vai trò đại diện.
  \end{block}
\end{frame}

\begin{frame}{Dự phòng --- Đe dọa đến tính hợp lệ}
  \begin{itemize}\small
    \item \textbf{Nội tại.} Xếp hạng RF/SHAP lấy từ tập huấn luyện gốc và giữ
          cố định qua các lần lấy mẫu lại --- nhất quán với thiết kế ``cấu hình
          cố định trước'', nhưng có thể gây rò rỉ nhẹ ở phần xếp hạng. Xếp hạng
          lại theo từng lần lấy mẫu chặt chẽ hơn nhưng rất tốn kém với SHAP.
          MLP không được gán trọng số lớp (Spark ML không hỗ trợ).
    \item \textbf{Ngoại tại.} CICIDS2017 (2017) có thể không phản ánh lưu lượng
          hiện tại; các bộ dữ liệu NIDS chuẩn đều có hạn chế chất lượng đã ghi
          nhận. Kết luận liên bộ dữ liệu dựa trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018. Chưa đánh giá trước lưu lượng đối kháng. Bản thân
          việc khử trùng lặp cũng làm thay đổi phân bố benign/attack.
    \item \textbf{Thống kê.} $N{=}6$ cho \emph{power} thấp: kiểm định hai phía
          liệt kê đầy đủ chặn $p$ ở mức $2/2^N$, nên $N{=}6$ chỉ vừa đủ chạm
          ngưỡng 0,05. Các lần lấy mẫu chồng lấn, vi phạm giả định độc lập của
          phân phối $t$ --- do đó Nadeau--Bengio là căn cứ chính. Tăng $N$ với
          cross-validation lồng nhau là hướng củng cố tiếp theo.
  \end{itemize}
\end{frame}

\begin{frame}{Dự phòng --- Giao thức đo hiệu năng ở pha biên}
  \begin{itemize}\small
    \item Script điều phối phía máy chủ khởi động các vai trò pipeline trên
          Jetson qua SSH, sinh tải, thu thập log từng nút.
    \item $\geq$3 lần lặp mỗi cấu hình; lưu lượng khởi động \emph{loại khỏi}
          cửa sổ đo, để JVM/JIT warmup không làm phồng độ trễ batch đầu.
    \item Mọi truy vấn chỉ số canh đúng cửa sổ tải, không dùng khoảng thời gian
          trượt phía sau.
    \item Phân vị độ trễ tính \emph{theo từng nút} trên mẫu thô từng luồng; p95
          mức cụm lấy \emph{thận trọng} theo nút tệ nhất.
    \item Độ trễ đầu cuối dựa trên đồng hồ đồng bộ NTP và \emph{không} bao gồm
          chi phí ghi cơ sở dữ liệu.
    \item Năng lượng: \texttt{tegrastats}, nền idle 30\,giây rồi đến cửa sổ tải;
          chế độ hai nút cộng công suất cả hai board.
  \end{itemize}
\end{frame}
