\section{I.\ ĐẶT VẤN ĐỀ VÀ MỤC TIÊU NGHIÊN CỨU}
% =====================================================================

\begin{frame}{Bối cảnh và ba khoảng trống nghiên cứu}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Vấn đề thực tiễn}
      \begin{itemize}\small
        \item Thiết bị IoT nhiều, tài nguyên ít, bảo mật yếu.
        \item Lưu lượng lớn $\rightarrow$ cần nền tảng \key{phân tán}.
        \item Cần phát hiện \key{ngay tại biên} để giảm độ trễ.
      \end{itemize}
      \vspace{0.5em}
      \textbf{Ba khoảng trống}
      \begin{enumerate}\small
        \item RF, SHAP, PCA chưa được so sánh trên \emph{cùng một} pipeline.
        \item Điểm số CICIDS2017 thường \bad{gần như hoàn hảo} --- dấu hiệu rò rỉ.
        \item IDS biên hầu như chỉ đo trên \bad{một thiết bị}.
      \end{enumerate}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{Cách tiếp cận: hai pha bổ trợ}
        \key{(A) Offline} --- chọn mô hình nào, bao nhiêu đặc trưng.\\[0.5em]
        \key{(B) Biên} --- mô hình đó có chạy thời gian thực được không?
      \end{exampleblock}
    \end{column}
  \end{columns}
  \note{Nhấn mạnh: hai pha trả lời hai câu hỏi khác nhau, không phải hai phương
  án cạnh tranh. Pha A dưới quy trình loại trừ rò rỉ. Ba câu hỏi nghiên cứu cụ
  thể ở slide ngay sau.}
\end{frame}

\begin{frame}{Mục tiêu và ba câu hỏi nghiên cứu}
  \textbf{Mục tiêu.} Xây dựng hệ thống IDS trên Apache Spark, đánh giá dưới quy
  trình \key{loại trừ rò rỉ dữ liệu}, và kiểm chứng khả năng chạy thời gian thực
  trên cụm thiết bị biên.
  \vspace{0.8em}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ1 --- Giảm chiều}
        \small Cắt hơn 60\% đặc trưng có giữ được hiệu năng không?
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ2 --- Học tổ hợp}
        \small Có hơn mô hình đơn tốt nhất một cách \emph{đáng kể} không?
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{RQ3 --- Khả thi ở biên}
        \small Chạy được thời gian thực trên cụm Jetson không?
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small
  \textbf{Phạm vi.} CICIDS2017, phân loại nhị phân, cụm 2 Jetson Orin Nano
  Super. Bộ dữ liệu \emph{không} có lưu lượng IoT đặc thù --- tính ``IoT'' nằm ở
  \key{kịch bản triển khai biên}.
  \note{Đọc chậm ba RQ; hội đồng sẽ đối chiếu với phần kết luận. RQ2 được kiểm
  chứng bằng kiểm định thống kê chứ không so một lần đo. Nêu chủ động giới hạn
  phạm vi IoT và tấn công đối kháng nằm ngoài phạm vi --- slide Q1 có bản đầy
  đủ kèm mô hình mối đe dọa.}
\end{frame}

\begin{frame}{Pha A --- Quy trình loại trừ rò rỉ và bốn chiến lược so sánh}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Hai nguồn rò rỉ, xử lý \emph{trước khi chia tập}}
      \begin{enumerate}\small
        \item \key{Đặc trưng rò rỉ nhãn}: \texttt{destination\_port} mã hóa
              thẳng dịch vụ bị tấn công.
        \item \key{Trùng lặp mức đặc trưng}: cùng một mẫu lọt vào cả tập huấn
              luyện lẫn tập kiểm tra.
      \end{enumerate}
      \vspace{0.4em}
      \small Nhóm \emph{nhãn xung đột} bị \bad{loại cả nhóm}: 270 nhóm,
      44.260 dòng. Còn 77 đặc trưng.
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Bốn chiến lược, cùng một pipeline}
      \begin{itemize}\small
        \item Baseline 77 đặc trưng $\cdot$ RF Top-$K$ $\cdot$ SHAP Top-$K$
              $\cdot$ PCA $k$=40.
        \item 8 thuật toán $+$ \key{Soft Voting}, \key{Hybrid Bagging}.
      \end{itemize}
      \vspace{0.4em}
      \small \textbf{Chống thiên lệch}: cấu hình cố định trước; tập kiểm tra
      \bad{không tham gia} khâu lựa chọn nào.
    \end{column}
  \end{columns}
  \note{Đóng góp phương pháp luận quan trọng nhất. Nói rõ vì sao khử trùng lặp
  tất định lại quan trọng: kết quả không phụ thuộc cách Spark phân vùng dữ liệu.
  Xếp hạng RF/SHAP chỉ dùng tập huấn luyện; chọn mô hình trên tập kiểm định tách
  từ tập huấn luyện. PCA dùng k=40 vì là phép trích xuất --- slide Q3 có lý do

\end{frame}

\begin{frame}{KQ1 --- Giảm 60\% đặc trưng mà vẫn giữ hiệu năng}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \centering\small
      \fitwidth{%
      \begin{tabular}{llccc}
        \toprule
        Phương pháp & Mô hình & F1 & Huấn luyện (s) & Đặc trưng \\
        \midrule
        Baseline    & XGBoost & 0,9971 & 4615,5 & 77 \\
        \textbf{SHAP Top-30} & XGBoost & \num{0,9970} & \num{1226,1} & \num{30} \\
        PCA $k$=40  & XGBoost & 0,9939 & 1444,8 & 40 \\
        RF Top-30   & XGBoost & 0,9922 & 1295,1 & 30 \\
        \bottomrule
      \end{tabular}
      }
      \vspace{0.6em}

      \small
      \fitwidth{%
      \begin{tabular}{lccc}
        \toprule
        Thuật toán & RF Top-30 & SHAP Top-30 & Chênh lệch t.đối \\
        \midrule
        Random Forest & 0,9879 & 0,9955 & \good{$+$0,77\%} \\
        XGBoost       & 0,9922 & 0,9970 & \good{$+$0,48\%} \\
        GBT           & 0,9844 & 0,9918 & \good{$+$0,75\%} \\
        Decision Tree & 0,9835 & 0,9948 & \good{$+$1,15\%} \\
        \bottomrule
      \end{tabular}
      }
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{itemize}
        \item SHAP Top-30 giữ F1 ngang baseline, cắt \good{60\% đặc trưng} và
              \good{73\% thời gian huấn luyện}.
        \item Cùng số đặc trưng, SHAP vượt RF Importance trên \emph{mọi} thuật
              toán họ cây.
        \item Học tổ hợp \emph{không} vượt được XGBoost đơn --- xem slide sau.
      \end{itemize}
    \end{column}
  \end{columns}
  \note{Trả lời RQ1 ngay tại đây, nhưng chưa kết luận vội --- còn phải kiểm
  định thống kê ở slide sau. Nói thêm: ít đặc trưng hơn thì chi phí
  VectorAssembler và bộ nhớ mô hình ở biên cũng nhỏ hơn; SHAP còn cho phép
  giải thích từng quyết định, có giá trị cho vận hành SOC.}
\end{frame}

\begin{frame}{KQ2 --- Học tổ hợp không vượt được mô hình đơn tốt nhất}
  \centering\small
  \fitwidth{%
  \begin{tabular}{lccc}
    \toprule
    Trên SHAP Top-30 (cấu hình triển khai) & F1 & Huấn luyện (s) & Dung lượng (MB) \\
    \midrule
    \textbf{XGBoost} (mô hình đơn) & \num{0,9970} & \num{1226} & --- \\
    Soft Voting                    & 0,9967 & --- & 0,02 \\
    Hybrid Bagging                 & 0,9966 & \bad{4909} & \bad{7,15} \\
    \bottomrule
  \end{tabular}
  }
  \par\vspace{0.5em}
  \raggedright
  \begin{itemize}
    \item Học tổ hợp chỉ thắng ở cấu hình mà mô hình đơn \emph{yếu}: RF Top-30
          (0,9938 so với 0,9922) và PCA (0,9940 so với 0,9939).
    \item Ở hai cấu hình mạnh nhất (Baseline, SHAP Top-30) thì \bad{thua} mô
          hình đơn --- nó \emph{bù cho bộ đặc trưng kém}, không tạo thêm năng lực.
  \end{itemize}
  \takeaway{Trả lời \key{RQ2}: không có lợi thế đáng kể. Với thiết bị biên, mô
  hình đơn thắng cả về F1 lẫn chi phí --- huấn luyện nhanh gấp 4 lần, mô hình
  nhỏ hơn nhiều.}
  \note{Đây là bằng chứng cho RQ2, và là một kết quả âm tính --- nói thẳng ra
  thay vì né. Hybrid Bagging ghép Top-3 thuật toán có F1 cao nhất trên tập kiểm
  định, có trọng số. Dung lượng XGBoost trên cụm phân tán báo thiếu (0,003 MB)
  nên để trống thay vì ghi con số sai. Slide sau sẽ cho thấy ngay cả chênh lệch
  giữa các cấu hình hàng đầu cũng không có ý nghĩa thực tiễn.}
\end{frame}

\begin{frame}{KQ3 --- Kiểm định thống kê thay vì so sánh một lần đo}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \centering\small
      \fitwidth{%

        \item \textbf{Triển khai}: cụm 2 Jetson Orin Nano Super (8\,GB) $+$ máy
              chủ điều phối.
      \end{itemize}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{alertblock}{Lưu ý trung thực về phạm vi ``IoT''}
        \footnotesize
        CICIDS2017 là lưu lượng doanh nghiệp \emph{mô phỏng}, không chứa giao
        thức đặc thù IoT (MQTT, CoAP). Tính chất ``IoT'' nằm ở \key{kịch bản
        triển khai}: toàn bộ pipeline suy luận được đo trên \emph{thiết bị biên
        tài nguyên hạn chế} --- lớp gateway trong kiến trúc IoT ba tầng --- chứ
        không ở bản chất lưu lượng huấn luyện.
      \end{alertblock}
      \begin{block}{Mô hình mối đe dọa}
        \footnotesize
        Kẻ tấn công ở bên trong hoặc ở biên mạng; phạm vi giới hạn ở các tấn
        công đã biết kiểu CICIDS2017. Tấn công đối kháng (adversarial/evasion)
        \bad{nằm ngoài phạm vi} đánh giá.
      \end{block}
    \end{column}
  \end{columns}
  \note{Chủ động nêu giới hạn phạm vi IoT trước khi hội đồng hỏi. Đây là câu
  hỏi phản biện gần như chắc chắn sẽ có.}
\end{frame}

\begin{frame}[label=bk-pca]{Q3 --- Vì sao PCA dùng $k$=40, và vì sao loại LightGBM?}
  \begin{block}{PCA $k$=40 thay vì $k$=30}
    \small
    PCA là phép \emph{trích xuất}, không phải \emph{chọn lọc}: mỗi thành phần
    chính là một tổ hợp tuyến tính chỉ mang một phần phương sai, nên cần nhiều
    thành phần hơn để giữ lượng thông tin tương đương tập 30 đặc trưng gốc.
    Trong dải quét chung $\{20,30,40\}$, $k$=40 là mức giữ phương sai cao nhất,
    nên là điểm vận hành đại diện \emph{công bằng nhất} cho PCA; ép $k$=30 sẽ
    làm PCA mất thêm phương sai và bất lợi khi so sánh.
  \end{block}
  \begin{block}{LightGBM bị loại khỏi so sánh}
    \small
    Thư viện native của LightGBM (qua SynapseML) chỉ hỗ trợ kiến trúc x86\_64,
    không chạy được trên thiết bị biên Jetson Orin Nano Super (ARM64) --- vốn là
    \emph{đích triển khai} của luận văn. Trong họ gradient boosting, XGBoost và
    GBT đóng vai trò đại diện.
  \end{block}
\end{frame}

\begin{frame}[label=bk-threats]{Q6 --- Các yếu tố ảnh hưởng tính hợp lệ}
  \begin{itemize}\small
    \item \textbf{Nội tại.} Xếp hạng RF/SHAP lấy từ tập huấn luyện gốc và giữ
          cố định qua các lần lấy mẫu lại --- nhất quán với thiết kế ``cấu hình
          cố định trước'', nhưng có thể gây rò rỉ nhẹ ở phần xếp hạng. Xếp hạng
          lại theo từng lần lấy mẫu chặt chẽ hơn nhưng rất tốn kém với SHAP.
          MLP không được gán trọng số lớp (Spark ML không hỗ trợ).
    \item \textbf{Ngoại tại.} CICIDS2017 (2017) có thể không phản ánh lưu lượng
          hiện tại; các bộ dữ liệu NIDS chuẩn đều có hạn chế chất lượng đã ghi
          nhận. Kết luận liên bộ dữ liệu dựa trên \emph{mẫu phân tầng} của
          CSE-CIC-IDS2018. Chưa đánh giá trước lưu lượng đối kháng. Bản thân
          việc khử trùng lặp cũng làm thay đổi phân bố benign/attack.
    \item \textbf{Thống kê.} $N{=}6$ cho \emph{power} thấp: kiểm định hai phía
          liệt kê đầy đủ chặn $p$ ở mức $2/2^N$, nên $N{=}6$ chỉ vừa đủ chạm
          ngưỡng 0,05. Các lần lấy mẫu chồng lấn, vi phạm giả định độc lập của
          phân phối $t$ --- do đó Nadeau--Bengio là căn cứ chính. Tăng $N$ với
          cross-validation lồng nhau là hướng củng cố tiếp theo.
  \end{itemize}
\end{frame}

\begin{frame}[label=bk-modes]{Q8 --- Ba chế độ triển khai và môi trường đo}

        \item[Đơn nút] toàn bộ pipeline trong một tiến trình --- mốc tham chiếu.

      luồng chỉ tính \emph{một lần}, tại nơi có kết luận: ở cổng nếu bị bỏ qua,
      ở bộ phân loại nếu được chuyển tiếp --- nhờ vậy luồng đi qua cả hai nút
      không bị đếm hai lần.

        Ampere $\approx$67 TOPS, 8\,GB LPDDR5, SSD NVMe 256\,GB.\\[0.3em]

        Tải: phát lại CICIDS2017 ở 100 luồng/giây.\\[0.3em]
        $\geq$3 lần lặp; lưu lượng khởi động \emph{loại khỏi} cửa sổ đo.\\[0.3em]
        Năng lượng: \texttt{tegrastats}, trừ nền idle.
      \end{block}
    \end{column}

\begin{frame}[label=bk-protocol]{Q9 --- Giao thức đo hiệu năng ở pha biên}
  \begin{itemize}\small

          Jetson qua SSH, sinh tải, thu thập log từng nút.
    \item $\geq$3 lần lặp mỗi cấu hình; lưu lượng khởi động \emph{loại khỏi}

    \item Mọi truy vấn chỉ số canh đúng cửa sổ tải, không dùng khoảng thời gian
          trượt phía sau.
    \item Phân vị độ trễ tính \emph{theo từng nút} trên mẫu thô từng luồng; p95
          mức cụm lấy \emph{thận trọng} theo nút tệ nhất.
    \item Độ trễ đầu cuối dựa trên đồng hồ đồng bộ NTP và \emph{không} bao gồm
          chi phí ghi cơ sở dữ liệu.
    \item Năng lượng: \texttt{tegrastats}, nền idle 30\,giây rồi đến cửa sổ tải;
          chế độ hai nút cộng công suất cả hai board.

\end{frame}

\begin{frame}[label=bk-future]{Q12 --- Hướng phát triển chi tiết}
  \begin{itemize}\small
    \item \textbf{Cải thiện độ nhạy cho lớp tấn công hiếm.} Xử lý mất cân bằng
          có chủ đích (undersampling/SMOTE áp dụng \emph{sau} khi chia tập),
          focal loss, hạ ngưỡng quyết định theo lớp; \key{kiến trúc phân tầng} ---
          tận dụng chính thiết kế cổng hai tầng: tầng nhị phân ở biên giữ recall
          tổng, còn bộ định danh loại chỉ huấn luyện trên \emph{luồng tấn công},
          huấn luyện lại ngoại tuyến mà không đụng hệ thống đang chạy; thêm
          \key{đặc trưng chuỗi thời gian theo host} cho beaconing C\&C.
    \item \textbf{Kiểm chứng trên dữ liệu IoT chuyên biệt.} Bot-IoT, TON\_IoT,
          CIC-IoT-2023; mở rộng thí nghiệm liên bộ dữ liệu lên toàn bộ
          CSE-CIC-IDS2018.
    \item \textbf{Tối ưu suy luận biên bằng ONNX/TensorRT.} Tích hợp phiên suy
          luận ONNX vào pipeline streaming, tăng tốc bằng TensorRT trên GPU
          Jetson; đo độ trễ đầu cuối đầy đủ.
    \item \textbf{Bền vững trước tấn công đối kháng.} Đặc biệt hai bề mặt tấn
          công mới do cổng tạo ra: giả dạng lưu lượng bình thường để vượt cổng,
          và cố ý tạo sai số tái tạo cao để dồn tải lên nút phân loại.
  \end{itemize}
  \note{Mỗi hướng phát triển ứng với một hạn chế ở slide trước --- đó là cách
  thể hiện sự làm chủ đề tài.}
\end{frame}
